name: WRF-SMN (modelo 4 km)

# Una vez por hora busca si el SMN publicó un ciclo nuevo de su modelo WRF (00, 06, 12 y 18 UTC).
# Si hay uno nuevo, lee el recorte de CABA y lo guarda en la rama "wrf". Las alertas lo toman de ahí.
on:
  schedule:
    - cron: "40 * * * *"
  workflow_dispatch:
    inputs:
      forzar:
        description: "Volver a procesar el último ciclo aunque ya esté"
        type: boolean
        default: false

permissions:
  contents: write

concurrency:
  group: wrf
  cancel-in-progress: false

jobs:
  wrf:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip
          cache-dependency-path: requirements-wrf.txt

      - run: pip install -r requirements-wrf.txt

      - name: Restaurar último WRF procesado
        run: |
          mkdir -p wrf
          if git ls-remote --exit-code --heads origin wrf > /dev/null; then
            git fetch --depth=1 origin wrf
            git archive FETCH_HEAD | tar -x -C wrf
          fi

      - name: Procesar ciclo nuevo del WRF-SMN
        run: python wrf.py ${{ inputs.forzar && '--forzar' || '' }}

      - name: Guardar en la rama wrf
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          [ -f wrf/wrf.json ] || { echo "Nada para guardar"; exit 0; }
          cd wrf
          git init -q -b wrf
          git config user.name "alertas-clima-caba[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A
          git commit -q -m "WRF $(python3 -c "import json;print(json.load(open('wrf.json'))['ciclo'])")"
          git push -q -f "https://x-access-token:${GH_TOKEN}@github.com/${GITHUB_REPOSITORY}.git" wrf
